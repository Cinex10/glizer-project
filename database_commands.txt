-- Voir les transactions échouées
SELECT * FROM bots_transactions WHERE status = 'failure';

-- Voir les transactions PUBG seulement
SELECT * FROM bots_transactions WHERE bot_type = 'pubg';

-- Voir les transactions Yalla Ludo seulement
SELECT * FROM bots_transactions WHERE bot_type = 'yalla_ludo';

-- Voir les transactions d'un bot spécifique
SELECT * FROM bots_transactions WHERE bot_num = 9;

-- Voir les transactions avec des critères multiples
SELECT * FROM bots_transactions 
WHERE bot_type = 'pubg' AND bot_num IN (9, 10) AND status = 'pending';

-- Compter les transactions par statut
SELECT status, COUNT(*) as count 
FROM bots_transactions 
GROUP BY status;

-- Compter les transactions par bot_type
SELECT bot_type, COUNT(*) as count 
FROM bots_transactions 
GROUP BY bot_type;

-- Voir les transactions récentes (par ID)
SELECT * FROM bots_transactions 
ORDER BY id DESC 
LIMIT 10;

-- Rechercher par player_id dans le payload
SELECT * FROM bots_transactions 
WHERE payload->>'player_id' = '533938203';

-- Rechercher par email dans le payload

WHERE payload->>'email' = 'player@example.com';

-- Voir la structure du payload d'une transaction
SELECT id, payload 
FROM bots_transactions

========================================
📊 REQUÊTES AVANCÉES
========================================

-- Statistiques par bot
SELECT 
    bot_num,
    bot_type,
    status,
    COUNT(*) as count
FROM bots_transactions 
GROUP BY bot_num, bot_type, status
ORDER BY bot_num, bot_type, status;

-- Taux de réussite par bot
SELECT 
    bot_num,
    bot_type,
    COUNT(*) as total,
    COUNT(CASE WHEN status = 'success' THEN 1 END) as successful,
    ROUND(
        COUNT(CASE WHEN status = 'success' THEN 1 END) * 100.0 / COUNT(*), 
        2
    ) as success_rate
FROM bots_transactions 
GROUP BY bot_num, bot_type
ORDER BY bot_num;

-- Transactions par heure (si vous avez des timestamps dans l'ID)
SELECT 
    DATE_TRUNC('hour', TO_TIMESTAMP(CAST(SPLIT_PART(id, '-', 2) AS BIGINT))) as hour,
    COUNT(*) as transaction_count
FROM bots_transactions 
WHERE id LIKE 'tx-%'
GROUP BY hour
ORDER BY hour DESC
LIMIT 24;

-- Top 10 des player_id les plus actifs
SELECT 
    payload->>'player_id' as player_id,
    COUNT(*) as transaction_count
FROM bots_transactions 
WHERE payload->>'player_id' IS NOT NULL
GROUP BY payload->>'player_id'
ORDER BY transaction_count DESC
LIMIT 10;

========================================
📋 INFORMATIONS UTILES
========================================

Structure de la table bots_transactions:
- id: VARCHAR (clé primaire)
- bot_num: INTEGER (1-10)
- status: VARCHAR ('pending', 'success', 'failure')
- payload: JSONB ({player_id, code, email, password})
- bot_type: VARCHAR ('yalla_ludo', 'pubg')

Contraintes:
- bot_num doit être entre 1 et 10
- bot_type doit être 'yalla_ludo' ou 'pubg'

Exemple de payload valide:
{
    "player_id": "533938203",
    "code": "TESTCODE123",
    "email": "player@example.com",
    "password": "playerpassword"
}
